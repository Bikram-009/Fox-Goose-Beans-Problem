01 — Problem Overview

The River Crossing Constraint Problem

A farmer must transport three entities across a river in a small boat. Only one can travel with the farmer at a time. Certain combinations left unattended cause irreversible conflict.

🦊🪿
Fox Eats Goose
If the fox and goose are left alone on the same bank without the farmer's supervision, the fox will eat the goose.
⚠ INVALID STATE
🪿🌱
Goose Eats Beans
If the goose and beans are left unsupervised on the same bank, the goose will consume the beans — another forbidden configuration.
⚠ INVALID STATE
🦊🌱
Fox + Beans = Safe
The fox has zero interest in beans. These two can coexist on any bank without supervision. This is the critical insight that unlocks the solution!
✓ VALID STATE
🚣
One Passenger Per Trip
The farmer can carry at most one item per crossing, or travel alone. Careful trip ordering is essential to reach [All Right] in minimum steps.

03 — Code & Algorithm

From Concept to Implementation

Study the pseudocode to understand the logic, then examine the working Java implementation. Both BFS and DFS are fully functional — copy and run them directly.

◈ BFS Pseudocode Breadth-First Search
// BFS: Optimal and Complete   FUNCTION BFS(startState, goalState): queue ← new Queue() // FIFO data structure visited ← new Set() // Tracks explored states parent ← new Map() // Reconstructs final path   queue.enqueue(startState) visited.add(startState) parent[startState] ← null   WHILE queue NOT EMPTY: current ← queue.dequeue()   IF current == goalState: RETURN buildPath(parent, current)   FOR EACH next IN getNextStates(current): IF next NOT IN visited AND isValid(next):     visited.add(next)     parent[next] ← current     queue.enqueue(next)   RETURN null // No path found   // Complexity: Time O(b^d), Space O(b^d) // Guarantees OPTIMAL (shortest) solution ✓
◈ DFS Pseudocode Depth-First Search
// DFS: Memory-efficient, Not Optimal   FUNCTION DFS(state, goal, visited, path): IF state == goal: path.add(state) RETURN true // Goal reached   IF state IN visited: RETURN false // Cycle detected, backtrack   visited.add(state) path.add(state)   FOR EACH next IN getNextStates(state): IF isValid(next): IF DFS(next, goal, visited, path):     RETURN true   // No valid path found — backtrack path.removeLast() RETURN false   // Complexity: Time O(b^m), Space O(b·m) // NOT guaranteed optimal ✗ (may miss shorter paths)
◈ Shared Helper Functions: isValid() & getNextStates()
// Check if a state breaks any puzzle rule FUNCTION isValid(state): farmer, fox, goose, beans ← state   // Fox eats Goose when unsupervised IF fox == goose AND fox ≠ farmer: RETURN false   // Goose eats Beans when unsupervised IF goose == beans AND goose ≠ farmer: RETURN false   RETURN true // State is safe
// Generate all possible next states FUNCTION getNextStates(state): nextStates ← []   // Option 1: Farmer crosses alone nextStates.add(flip(state, FARMER))   // Options 2–4: Farmer takes one entity FOR EACH entity IN [FOX, GOOSE, BEANS]: IF entity.side == farmer.side: nextStates.add(   flip(state, FARMER, entity))   RETURN nextStates
FoxGooseBeansBFS.java 
import java.util.*;

/**
 * BFS Solver for Fox, Goose and Beans puzzle.
 * State = 4-bit int: bit3=Farmer, bit2=Fox, bit1=Goose, bit0=Beans
 * 0 = Left bank, 1 = Right bank
 */
public class FoxGooseBeansBFS {

    static final int START = 0b0000, GOAL = 0b1111;

    // Returns false if the state is unsafe (unattended predator-prey pair)
    static boolean isValid(int s) {
        int f=(s>>3)&1, x=(s>>2)&1, g=(s>>1)&1, b=s&1;
        return !(x==g && x!=f) && !(g==b && g!=f);
    }

    // All 4 possible moves: farmer alone, or with fox/goose/beans
    static List<Integer> nextStates(int cur) {
        List<Integer> res = new ArrayList<>();
        for (int m : new int[]{8,12,10,9})  // XOR masks for each move
            if (isValid(cur ^ m)) res.add(cur ^ m);
        return res;
    }

    // BFS: explores states level-by-level, guarantees shortest path
    public static List<Integer> solve() {
        Queue<Integer> q = new LinkedList<>();
        Map<Integer,Integer> par = new HashMap<>();
        q.add(START); par.put(START, -1);

        while (!q.isEmpty()) {
            int cur = q.poll();
            if (cur == GOAL) return buildPath(par, cur);
            for (int next : nextStates(cur))
                if (!par.containsKey(next)) {
                    par.put(next, cur);
                    q.add(next);
                }
        }
        return Collections.emptyList();
    }

    // Walks the parent map backwards to reconstruct the solution path
    static List<Integer> buildPath(Map<Integer,Integer> par, int goal) {
        List<Integer> path = new ArrayList<>();
        for (int s = goal; s != -1; s = par.get(s))
            path.add(0, s);
        return path;
    }

    public static void main(String[] args) {
        solve().forEach(s ->
            System.out.printf("[F=%d X=%d G=%d B=%d]%n",
                (s>>3)&1, (s>>2)&1, (s>>1)&1, s&1));
    }
}
FoxGooseBeansDFS.java 
import java.util.*;

/**
 * DFS Solver for Fox, Goose and Beans puzzle.
 * Uses same 4-bit state encoding as BFS version.
 * DFS is NOT guaranteed to find the shortest path.
 */
public class FoxGooseBeansDFS {

    static final int START = 0b0000, GOAL = 0b1111;

    // Reuse isValid and nextStates from BFS (same puzzle, same rules)
    static boolean isValid(int s) {
        int f=(s>>3)&1, x=(s>>2)&1, g=(s>>1)&1, b=s&1;
        return !(x==g && x!=f) && !(g==b && g!=f);
    }

    // Recursive DFS: dives deep before backtracking
    static boolean dfs(int state, Set<Integer> visited, List<Integer> path) {
        if (state == GOAL)          { path.add(state); return true;  }  // Goal reached
        if (visited.contains(state)) return false;              // Cycle — backtrack

        visited.add(state);
        path.add(state);

        for (int m : new int[]{8,12,10,9}) {  // Try all 4 moves
            int next = state ^ m;
            if (isValid(next) && dfs(next, visited, path)) return true;
        }

        path.remove(path.size() - 1);  // Backtrack: undo this step
        return false;
    }

    public static void main(String[] args) {
        List<Integer> path    = new ArrayList<>();
        Set<Integer>  visited = new HashSet<>();
        dfs(START, visited, path);
        path.forEach(s ->
            System.out.printf("[F=%d X=%d G=%d B=%d]%n",
                (s>>3)&1, (s>>2)&1, (s>>1)&1, s&1));
    }
}
